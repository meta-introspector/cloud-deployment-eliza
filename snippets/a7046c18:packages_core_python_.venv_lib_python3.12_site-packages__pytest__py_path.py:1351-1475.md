def copychunked(src, dest):

    fsrc = src.open("rb")
    try:
        fdest = dest.open("wb")
        try:
            while 1:
                buf = fsrc.read(chunksize)
                if not buf:

                fdest.write(buf)
        finally:

    finally:
        fsrc.close()
