        return 0;
    }

    if (overallocate) {
#ifdef MS_WINDOWS

        if (size <= (PY_SSIZE_T_MAX - size / 2)) {
            size += size / 2;
        }
#else

        if (size <= (PY_SSIZE_T_MAX - size / 4)) {
            size += size / 4;
        }
#endif

        if (_PyBytes_Resize(&writer->obj, size)) {
            return -1;
