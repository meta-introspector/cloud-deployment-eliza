    def iter_tokens(self) -> Iterable[str]:
        """Generate tokens for this node."""

            yield self.key_separator

                yield self.open_brace

                    yield ","
                else:

                        yield from child.iter_tokens()
                        if not child.last:
                            yield self.separator
