    """

    name = 'MCSchema'
    url = 'https://learn.microsoft.com/en-us/minecraft/creator/reference/content/schemasreference/'
    aliases = ['mcschema']
    filenames = ['*.mcschema']
    mimetypes = ['text/mcschema']
    version_added = '2.14'

    tokens = {
        'commentsandwhitespace': [
            (r'\s+', Whitespace),

            include('commentsandwhitespace'),
