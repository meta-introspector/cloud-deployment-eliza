    def _pprint_bytes(
        self,
        object: Any,
        stream: IO[str],
        indent: int,
        allowance: int,
        context: set[int],
        level: int,
    ) -> None:
        write = stream.write
        if len(object) <= 4:
            write(repr(object))
            return
        parens = level == 1

            indent += 1
            allowance += 1
            write("(")

        for rep in _wrap_bytes_repr(object, self._width - indent, allowance):

            write(rep)

            write(")")
