    cdef bytes desc = cparser.llhttp_get_error_reason(parser)

    if errno in {cparser.HPE_CB_MESSAGE_BEGIN,

                 cparser.HPE_INVALID_CHUNK_SIZE,

                 cparser.HPE_INVALID_TRANSFER_ENCODING}:
