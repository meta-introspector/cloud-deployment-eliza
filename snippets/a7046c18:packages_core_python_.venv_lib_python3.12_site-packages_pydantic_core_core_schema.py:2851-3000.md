    """

    ```py
    from pydantic_core import SchemaValidator, ValidationError, core_schema

        core_schema.json_or_python_schema(
    ```

        type='json-or-python',
        json_schema=json_schema,
        python_schema=python_schema,
        ref=ref,
        metadata=metadata,
        serialization=serialization,
    )
