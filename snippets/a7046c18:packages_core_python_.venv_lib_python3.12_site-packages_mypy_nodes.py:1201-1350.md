        self.invalid_partial_type = False

    @property
    def name(self) -> str:
        return self._name

    @property
    def fullname(self) -> str:
        return self._fullname

    def __repr__(self) -> str:
        name = self.fullname or self.name
        return f"<Var {name!r} at {hex(id(self))}>"

    def accept(self, visitor: NodeVisitor[T]) -> T:
