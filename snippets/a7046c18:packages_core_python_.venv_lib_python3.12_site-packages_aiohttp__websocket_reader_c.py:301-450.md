        if self._tail:

                start_pos += 2

                fin = (first_byte >> 7) & 1

                rsv2 = (first_byte >> 5) & 1
                rsv3 = (first_byte >> 4) & 1

                # frame-rsv3 = %x0 ;

                        WSCloseCode.PROTOCOL_ERROR,
                        "Received fragmented control frame",
